nhd_flowline <- readRDS(here::here("data-raw", "shapes", "nhd_flowline_hr.rds"))

flowlines <- nhd_flowline |>
  st_set_geometry("shape") |> 
  st_intersection(klamath_hucs |> st_union())
## Warning: attribute variables are assumed to be spatially constant throughout
## all geometries
names(flowlines)[names(flowlines) == "shape"] <- "geometry"
st_geometry(flowlines) <- "geometry"
flow_net <- flowlines 
seed_comid <- 50000400214915
seed_row <- flow_net |>
  st_drop_geometry() |>
  filter(comid == seed_comid)

edges <- flow_net |>
  st_drop_geometry() |>
  select(comid, from_node, to_node) |> 
  filter(!is.na(to_node))

g <- graph_from_data_frame(
  d = edges |> select(from_node, to_node),
  directed = TRUE
)

comp <- components(g, mode = "weak")

node_membership <- tibble(
  node = names(comp$membership),
  component_id = as.integer(comp$membership)
)

flow_comp <- flow_net |>
  left_join(
    node_membership |> 
      rename(from_node = node) |> 
      mutate(from_node = as.numeric(from_node)),
    by = "from_node"
  )

seed_component <- flow_comp |>
  st_drop_geometry() |>
  filter(comid == seed_comid) |>
  pull(component_id)

flow_clean <- flow_comp |>
  filter(component_id == seed_component)

flow_filter <- flow_clean |> 
  filter(stream_orde >= 5) |> 
  filter(tot_da_sq_km >= 1)
